## 1. Architecture Overview

### Unity (Current) → Babylon.js (Target)

| Unity Concept | Babylon.js Equivalent |
|---|---|
| `MonoBehaviour` | Custom class with `scene.onBeforeRenderObservable` |
| `GameObject` + Components | `TransformNode` / `Mesh` + attached behaviors |
| `[SerializeField]` properties | Constructor config objects / JSON configs |
| `Update()` loop | `scene.registerBeforeRender()` or `scene.onBeforeRenderObservable` |
| `CharacterController` | Custom physics or `PhysicsAggregate` with Havok/Cannon |
| `NavMeshAgent` | Recast.js navigation plugin (`RecastJSPlugin`) |
| `Animator` + `CrossFade` | `AnimationGroup.start()` with blending weights |
| `Physics.OverlapBox` | `PhysicsEngine.raycast` + custom AABB checks |
| `Rigidbody.AddForce` | `PhysicsBody.applyImpulse()` |
| `Instantiate(prefab)` | `mesh.clone()` or `mesh.createInstance()` |
| `ScriptableObject` | Plain TypeScript interfaces / JSON data files |
| `Coroutine` / `Invoke` | `setTimeout`, `setInterval`, or `Observable`-based timers |
| `Input.GetKey` | `scene.onKeyboardObservable` / `ActionManager` |
| `EventManager` (static bus) | Custom `EventEmitter` or RxJS `Subject` |
| `Destroy(obj, delay)` | `setTimeout(() => mesh.dispose(), delay)` |

### Tech Stack

```
Frontend:  React 18 + TypeScript + Vite
Engine:    Babylon.js 7.x (WebGL2 / WebGPU)
Physics:   Havok Physics or Cannon.js
Nav Mesh:  Recast.js plugin
Audio:     Babylon.js Sound API
Backend:   Express.js (leaderboards, saves)
```

## 3. Engine Setup — BabylonEngine.ts

### C# Source Reference: `GameManager.cs`, Scene setup

The Unity project uses a singleton `GameManager` with `GameState` enum and references to all subsystems. In Babylon.js this becomes the engine bootstrap.

```typescript
// BabylonEngine.ts
import {
  Engine, Scene, FreeCamera, HemisphericLight, DirectionalLight,
  Vector3, Color3, Color4, PointerEventTypes, KeyboardEventTypes,
  GlowLayer, DefaultRenderingPipeline
} from "@babylonjs/core";

export class BabylonEngine {
  engine: Engine;
  scene: Scene;
  camera: FreeCamera;

  constructor(canvas: HTMLCanvasElement) {
    // Engine with WebGPU fallback to WebGL2
    this.engine = new Engine(canvas, true, {
      preserveDrawingBuffer: true,
      stencil: true,
      antialias: true
    });

    this.scene = new Scene(this.engine);
    this.scene.clearColor = new Color4(0.02, 0.02, 0.08, 1); // Dark sci-fi sky
    this.scene.collisionsEnabled = true;
    this.scene.gravity = new Vector3(0, -9.81, 0);

    // FPS Camera (replaces Unity's CameraController + PlayerCamera)
    this.camera = new FreeCamera("playerCam", new Vector3(0, 5, -10), this.scene);
    this.camera.attachControl(canvas, true);
    this.camera.applyGravity = true;
    this.camera.checkCollisions = true;
    this.camera.ellipsoid = new Vector3(0.5, 1, 0.5);
    this.camera.minZ = 0.1;
    this.camera.speed = 0.6;       // ← from moveSpeed = 6f
    this.camera.angularSensibility = 2000;

    // Pointer lock (replaces Unity's Cursor.lockState)
    this.scene.onPointerObservable.add((pointerInfo) => {
      if (pointerInfo.type === PointerEventTypes.POINTERDOWN) {
        canvas.requestPointerLock();
      }
    });

    this.setupLighting();
    this.setupPostProcessing();
    this.startRenderLoop();
  }

  private setupLighting(): void {
    // Ambient (replaces Unity's RenderSettings.ambientLight)
    const ambient = new HemisphericLight("ambient", new Vector3(0, 1, 0), this.scene);
    ambient.intensity = 0.3;
    ambient.diffuse = new Color3(0.6, 0.6, 0.8);

    // Directional sun
    const sun = new DirectionalLight("sun", new Vector3(-1, -2, -1), this.scene);
    sun.intensity = 0.8;

    // Neon glow layer (replaces Unity's bloom for neon signs)
    const glowLayer = new GlowLayer("glow", this.scene);
    glowLayer.intensity = 0.8;
  }

  private setupPostProcessing(): void {
    // DefaultRenderingPipeline replaces Unity's URP post-process stack
    const pipeline = new DefaultRenderingPipeline("pipeline", true, this.scene, [this.camera]);
    pipeline.bloomEnabled = true;
    pipeline.bloomThreshold = 0.7;
    pipeline.bloomWeight = 0.3;
    pipeline.chromaticAberrationEnabled = true;
    pipeline.chromaticAberration.aberrationAmount = 30;
    pipeline.fxaaEnabled = true;
    pipeline.imageProcessing.toneMappingEnabled = true;
  }

  private startRenderLoop(): void {
    this.engine.runRenderLoop(() => {
      this.scene.render();
    });
    window.addEventListener("resize", () => this.engine.resize());
  }

  dispose(): void {
    this.scene.dispose();
    this.engine.dispose();
  }
}
```

**Key migration notes:**
- Unity's `Time.deltaTime` → `scene.getEngine().getDeltaTime() / 1000` (Babylon returns ms)
- Unity's `Update()` → `scene.registerBeforeRender(callback)`
- Unity's `FixedUpdate()` → Use physics step callbacks if using Havok

## 4. Player Controller — PlayerController.ts

### C# Source Reference: `PlayerController.cs` (590 lines)

The Unity player uses `CharacterController.Move()` with manual gravity, dodge (i-frames), parry, lock-on targeting, jetpack, and solar-system gravity reorientation.

```typescript
// PlayerController.ts
import { Scene, FreeCamera, Vector3, KeyboardEventTypes, Scalar } from "@babylonjs/core";

interface PlayerConfig {
  moveSpeed: number;      // Unity: 6f
  sprintSpeed: number;    // Unity: 9f
  jumpForce: number;      // Unity: 20f
  gravity: number;        // Unity: -9.81f
  dodgeSpeed: number;     // Unity: 15f
  dodgeDuration: number;  // Unity: 0.3f
  dodgeCooldown: number;  // Unity: 1f
  parryWindow: number;    // Unity: 0.2f
  parryCooldown: number;  // Unity: 1f
  parryInvincibilityDuration: number; // Unity: 0.5f
  jetpackForce: number;   // Unity: 10f
  jetpackDuration: number; // Unity: 2f
}

const DEFAULT_CONFIG: PlayerConfig = {
  moveSpeed: 6,
  sprintSpeed: 9,
  jumpForce: 20,
  gravity: -9.81,
  dodgeSpeed: 15,
  dodgeDuration: 0.3,
  dodgeCooldown: 1,
  parryWindow: 0.2,
  parryCooldown: 1,
  parryInvincibilityDuration: 0.5,
  jetpackForce: 10,
  jetpackDuration: 2
};

export class PlayerController {
  private scene: Scene;
  private camera: FreeCamera;
  private config: PlayerConfig;

  // Movement state (mirrors Unity's private fields)
  private velocity = Vector3.Zero();
  private gravityVelocity = Vector3.Zero();
  private externalVelocity = Vector3.Zero();
  private isGrounded = false;
  private isSprinting = false;

  // Dodge state
  private isDodging = false;
  private dodgeTimer = 0;
  private dodgeCooldownTimer = 0;
  private dodgeDirection = Vector3.Zero();

  // Parry state
  private isParrying = false;
  private parryTimer = 0;
  private parryCooldownTimer = 0;

  // Input map
  private keys: Record<string, boolean> = {};

  constructor(scene: Scene, camera: FreeCamera, config?: Partial<PlayerConfig>) {
    this.scene = scene;
    this.camera = camera;
    this.config = { ...DEFAULT_CONFIG, ...config };

    this.setupInput();
    this.scene.registerBeforeRender(() => this.update());
  }

  /** Mirrors Unity Awake() — input event subscription */
  private setupInput(): void {
    this.scene.onKeyboardObservable.add((kbInfo) => {
      const key = kbInfo.event.key.toLowerCase();
      this.keys[key] = kbInfo.type === KeyboardEventTypes.KEYDOWN;

      // Dodge on Shift press (mirrors inputManager.OnDodgePressed)
      if (key === "shift" && kbInfo.type === KeyboardEventTypes.KEYDOWN) {
        this.onDodge();
      }
    });
  }

  /** Mirrors Unity Update() — sequential system calls */
  private update(): void {
    const dt = this.scene.getEngine().getDeltaTime() / 1000;
    this.handleGroundCheck();
    this.handleMovement(dt);
    this.handleDodge(dt);
    this.handleParry(dt);
    this.handleGravity(dt);
    this.applyMovement(dt);
  }

  /**
   * Mirrors HandleGroundCheck()
   * Unity: characterController.isGrounded
   * Babylon: Ray cast downward from player position
   */
  private handleGroundCheck(): void {
    const ray = new BABYLON.Ray(this.camera.position, Vector3.Down(), 1.2);
    const hit = this.scene.pickWithRay(ray);
    this.isGrounded = hit?.hit ?? false;
  }

  /**
   * Mirrors HandleMovement()
   * Unity: Camera-relative WASD with sprint toggle
   */
  private handleMovement(dt: number): void {
    if (this.isDodging) return;

    const forward = this.camera.getDirection(Vector3.Forward());
    const right = this.camera.getDirection(Vector3.Right());

    // Flatten to horizontal (mirrors Unity's Vector3.Project removal)
    forward.y = 0; forward.normalize();
    right.y = 0; right.normalize();

    let inputX = 0, inputZ = 0;
    if (this.keys["w"]) inputZ += 1;
    if (this.keys["s"]) inputZ -= 1;
    if (this.keys["a"]) inputX -= 1;
    if (this.keys["d"]) inputX += 1;

    this.isSprinting = this.keys["shift"] ?? false;
    const speed = this.isSprinting ? this.config.sprintSpeed : this.config.moveSpeed;

    const direction = forward.scale(inputZ).add(right.scale(inputX));
    if (direction.length() > 0.1) direction.normalize();

    this.velocity = direction.scale(speed);
  }

  /**
   * Mirrors HandleDodge() + OnDodge()
   * Provides i-frames via PlayerStats.SetInvincible
   */
  private onDodge(): void {
    if (this.dodgeCooldownTimer > 0) return;

    this.isDodging = true;
    this.dodgeTimer = this.config.dodgeDuration;
    this.dodgeCooldownTimer = this.config.dodgeCooldown;

    // Dodge in movement direction or forward
    this.dodgeDirection = this.velocity.length() > 0.1
      ? this.velocity.normalize()
      : this.camera.getDirection(Vector3.Forward());

    // i-frames: set invincible on PlayerStats
    // playerStats.setInvincible(true);
    // setTimeout(() => playerStats.setInvincible(false), this.config.dodgeDuration * 1000);
  }

  private handleDodge(dt: number): void {
    if (this.isDodging) {
      this.dodgeTimer -= dt;
      if (this.dodgeTimer <= 0) {
        this.isDodging = false;
      }
    }
    if (this.dodgeCooldownTimer > 0) {
      this.dodgeCooldownTimer -= dt;
    }
  }

  /**
   * Mirrors HandleParry() + OnParry()
   * i-frames with parryWindow timing
   */
  private handleParry(dt: number): void {
    if (this.isParrying) {
      this.parryTimer -= dt;
      if (this.parryTimer <= 0) this.isParrying = false;
    }
    if (this.parryCooldownTimer > 0) {
      this.parryCooldownTimer -= dt;
    }
  }

  /**
   * Mirrors HandleGravity()
   * Applies gravity when not grounded, jump impulse when grounded + space
   */
  private handleGravity(dt: number): void {
    if (this.isGrounded) {
      if (this.keys[" "]) {
        this.gravityVelocity = new Vector3(0, this.config.jumpForce, 0);
      } else {
        this.gravityVelocity = new Vector3(0, -2, 0); // Stick to ground
      }
    } else {
      this.gravityVelocity.y += this.config.gravity * dt;
    }
  }

  /**
   * Mirrors ApplyMovement()
   * Unity: characterController.Move(totalVelocity * dt)
   * Babylon: camera.position += totalVelocity * dt (with collision)
   */
  private applyMovement(dt: number): void {
    const horizontalVel = this.isDodging
      ? this.dodgeDirection.scale(this.config.dodgeSpeed)
      : this.velocity.add(this.externalVelocity);

    const totalVelocity = horizontalVel.add(this.gravityVelocity);
    this.camera.position.addInPlace(totalVelocity.scale(dt));

    // Damp external velocity (mirrors externalVelocityDamp)
    if (this.externalVelocity.length() > 0.01) {
      this.externalVelocity = Vector3.Lerp(this.externalVelocity, Vector3.Zero(), 5 * dt);
    }
  }

  /**
   * Mirrors ApplyKnockback(Vector3 force)
   * Called from DamageSystem when player takes a hit
   */
  applyKnockback(force: Vector3): void {
    this.externalVelocity.addInPlace(force);
    if (this.isDodging) {
      this.isDodging = false;
      this.dodgeTimer = 0;
    }
  }
}
```

**Migration notes for `PlayerController.cs`:**
- Unity's `CharacterController.Move()` handles collision internally; in Babylon.js use `camera.checkCollisions = true` or implement a custom collision sweep
- Lock-on targeting (`FindNearestEnemy`) maps to iterating `scene.getMeshesByTags("enemy")` and sorting by distance
- Camera reparenting for vehicles (`AttachCameraTo`/`ExitFromSpaceship`) maps to `camera.parent = targetNode`

## 5. State Machine — StateMachine.ts

### C# Source Reference: `Starfall.Core.StateMachine` — `IState.cs`, `BaseState.cs`, `StateMachine.cs`, `StateContainer.cs`

The Unity project features a sophisticated generic state machine with history tracking (up to 10 states), `GoBack()` support, and typed state containers.

```typescript
// StateMachine.ts
export interface IState {
  onBecameActive(): void;
  onBecameInactive(): void;
  update(dt: number): void;
  lateUpdate?(dt: number): void;
  parameterForEnteringState?: any;
}

export class StateMachine {
  private currentState: IState | null = null;
  private stateHistory: IState[] = [];
  private static MAX_HISTORY = 10; // Mirrors Unity's MaxHistorySize

  /** Mirrors SwitchState() — deactivate old, push history, activate new */
  switchState(newState: IState): void {
    if (this.currentState) {
      this.currentState.onBecameInactive();
      this.stateHistory.push(this.currentState);
      if (this.stateHistory.length > StateMachine.MAX_HISTORY) {
        this.stateHistory.shift();
      }
    }
    this.currentState = newState;
    this.currentState.onBecameActive();
  }

  /** Mirrors SwitchStateWithParameter() */
  switchStateWithParameter(newState: IState, param: any): void {
    newState.parameterForEnteringState = param;
    this.switchState(newState);
  }

  /** Mirrors GoBack() — pop history, restore previous state */
  goBack(): boolean {
    if (this.stateHistory.length === 0) return false;
    const previousState = this.stateHistory.pop()!;
    if (this.currentState) this.currentState.onBecameInactive();
    this.currentState = previousState;
    this.currentState.onBecameActive();
    return true;
  }

  update(dt: number): void { this.currentState?.update(dt); }
  lateUpdate(dt: number): void { this.currentState?.lateUpdate?.(dt); }
  getCurrentState(): IState | null { return this.currentState; }
}

/**
 * Typed state container — mirrors StateContainer<T>
 * Lookup by constructor type
 */
export class StateContainer<T extends IState> {
  private states: Map<Function, T> = new Map();

  addState(state: T): void {
    this.states.set(state.constructor, state);
  }

  getState<S extends T>(ctor: new (...args: any[]) => S): S | undefined {
    return this.states.get(ctor) as S | undefined;
  }
}
```

## 6. Combat & Combo System — ComboSystem.ts

### C# Source Reference: `ComboSystem.cs` (230 lines)

The Unity combo system features:
- **Combo chains** with indexed attack progression (`ComboChain.currentAttackIndex`)
- **Input buffering** during active attacks (`inputBufferTime = 0.2f`)
- **Timed combo windows** — must press attack within `comboWindow` seconds to continue
- **Auto-reset** after `resetTime = 1.5f` of inactivity
- **Physics hitboxes** via `Physics.OverlapBox` with knockback

```typescript
// ComboSystem.ts

/** Mirrors AttackData class */
interface AttackData {
  animationName: string;
  damage: number;          // Default: 15
  knockback: number;       // Default: 3
  attackDuration: number;  // Default: 0.6s
  comboWindow: number;     // Default: 0.3s (0 = last hit)
  hitboxSize: { x: number; y: number; z: number };  // Default: 1×1×2
  hitboxOffset: { x: number; y: number; z: number }; // Default: 0×1×1
}

/** Mirrors ComboChain class */
interface ComboChain {
  comboName: string;
  attacks: AttackData[];
  resetTime: number;      // Default: 1.5s
  currentAttackIndex: number;
  lastAttackTime: number;
  canCombo: boolean;
}

export class ComboSystem {
  private lightAttackCombo: ComboChain;
  private heavyAttackCombo: ComboChain;
  private currentCombo: ComboChain | null = null;
  private isAttacking = false;

      { animationName: "Attack1", damage: 15, knockback: 3,
        attackDuration: 0.6, comboWindow: 0.3,
        hitboxSize: { x: 1, y: 1, z: 2 }, hitboxOffset: { x: 0, y: 1, z: 1 } },
      { animationName: "Attack2", damage: 20, knockback: 4,
        attackDuration: 0.7, comboWindow: 0.3,
        hitboxSize: { x: 1, y: 1, z: 2 }, hitboxOffset: { x: 0, y: 1, z: 1 } },
      { animationName: "Attack3", damage: 30, knockback: 6,
        attackDuration: 0.8, comboWindow: 0,    // Last attack — no continuation
        hitboxSize: { x: 1.5, y: 1, z: 2.5 }, hitboxOffset: { x: 0, y: 1, z: 1.2 } },
    ]);

    this.heavyAttackCombo = this.createDefaultCombo("Heavy Combo", [
      { animationName: "HeavyAttack1", damage: 25, knockback: 6,
        attackDuration: 0.8, comboWindow: 0.4,
        hitboxSize: { x: 1.5, y: 1.5, z: 2.5 }, hitboxOffset: { x: 0, y: 1, z: 1.5 } },
      { animationName: "HeavyAttack2", damage: 40, knockback: 8,
        attackDuration: 1.0, comboWindow: 0,
        hitboxSize: { x: 2, y: 1.5, z: 3 }, hitboxOffset: { x: 0, y: 1, z: 1.5 } },
    ]);
  }

  /**
   * Mirrors OnLightAttack()
   * Three-path logic: start new combo, continue existing, or buffer input
   */
  onLightAttack(canAttack: boolean): void {
    if (canAttack) {
      this.startAttack(this.lightAttackCombo);
    } else if (this.currentCombo?.canCombo) {
      this.continueCombo();
    } else {
      this.bufferInput();
    }
  }

  /**
   * Mirrors ExecuteAttack(AttackData)
   * Schedules hitbox activation, combo window, and attack end
   */
  private executeAttack(attack: AttackData): void {
    this.isAttacking = true;

    // Hitbox activation after 200ms (mirrors WaitForSeconds(0.2f))
    setTimeout(() => this.activateHitbox(attack), 200);

    // Enable combo window (mirrors Invoke(EnableComboWindow))
    setTimeout(() => {
      if (this.currentCombo) this.currentCombo.canCombo = true;
    }, attack.comboWindow * 1000);

    // End attack after duration (mirrors Invoke(EndAttack))
    setTimeout(() => this.endAttack(), attack.attackDuration * 1000);
  }

  /**
   * Mirrors ActivateHitbox coroutine
   * Unity: Physics.OverlapBox → Babylon: custom AABB intersection test
   */
  private activateHitbox(attack: AttackData): void {
    // In Babylon.js, get all enemy meshes and check AABB overlap
    // const enemies = scene.getMeshesByTags("enemy");
    // For each enemy, check if within hitbox bounds relative to player
    // Apply damage and knockback via DamageSystem
  }

  update(dt: number): void {
    // Mirrors UpdateComboTimers() — auto-reset after resetTime
    if (this.currentCombo) {
      const elapsed = performance.now() / 1000 - this.currentCombo.lastAttackTime;
      if (elapsed > this.currentCombo.resetTime) {
        this.resetCombo();
      }
    }

    // Mirrors UpdateInputBuffer()
    if (this.inputBufferTimer > 0) {
      this.inputBufferTimer -= dt;
```

```typescript
   * Spawn a new projectile
   * Mirrors Projectile.Initialize(shooter, target)

    mesh.position = origin.clone();
    const mat = new StandardMaterial("projMat", this.scene);
    mat.emissiveColor = new Color3(0, 1, 1); // Cyan glow

      mesh, direction: direction.normalize(), config,
      target, currentSpeed: config.speed,
      spawnTime: performance.now() / 1000
    });

    const now = performance.now() / 1000;

        continue;

        p.direction = Vector3.Lerp(p.direction, toTarget, p.config.rotationSpeed * dt);
        p.direction.normalize();

        p.currentSpeed + p.config.acceleration * dt,
```

```typescript
export function applyDamage(target: IDamageable, info: DamageInfo): DamageResult {
  if (!target.isAlive || target.isInvulnerable) {
    return { damageAmount: 0, wasKilled: false };
  }

  // Apply resistance (mirrors DamageResistance lookup)
  let finalDamage = info.amount;
  const resistance = target.resistances.find(r => r.damageType === info.damageType);
  if (resistance) {
    finalDamage *= (1 - resistance.resistancePercent);
  }

  finalDamage = Math.max(1, finalDamage);
  target.health = Math.max(0, target.health - finalDamage);

  const result: DamageResult = {
    damageAmount: finalDamage,
    wasKilled: target.health <= 0
  };

  DamageEvents.emitDamaged(target, info, result);
  target.onDamaged?.(info, result);

  if (result.wasKilled) {
    DamageEvents.emitDeath(target);
    target.onDeath?.();
  }

  return result;
}
```

```typescript
      let roll = Math.random() * 100;
      // Diminishing rolls: keep generating while roll < chance, halve each time
      while (roll < cat.chance) {
        this.items.push(this.generateItem(cat.name, playerLevel));
        cat.chance /= 2; // Diminishing returns
        roll = Math.random() * 100;
      }
    }

    // Gold drop
    if (table.maxGold > 0) {
      const gold = Math.floor(
        table.minGold + Math.random() * (table.maxGold - table.minGold)
      );
      if (gold > 0) this.items.push({ name: "Credits", type: "currency", quantity: gold });
    }
  }

  /** Mirrors LootContainer.TakeItem / TakeAll */
  takeItem(index: number): GeneratedItem | null {
    if (index < 0 || index >= this.items.length) return null;
    return this.items.splice(index, 1)[0];
  }

  takeAll(): GeneratedItem[] {
    const all = [...this.items];
    this.items = [];
    return all;
  }
}
```

```typescript
export class TerrainSystem {

  /**
   * Mirrors ProceduralTerrainGenerator.GenerateHeightmap()
   * Chunk-based generation with caching
   */
  generateChunk(chunkX: number, chunkZ: number, params: TerrainParams): Mesh {
    const key = `${chunkX}_${chunkZ}`;

    if (!this.heightmapCache.has(key)) {
      const heightmap = this.generateHeightmap(chunkX, chunkZ, params);
      this.heightmapCache.set(key, heightmap);
    }

    return this.createMeshFromHeightmap(this.heightmapCache.get(key)!, params);
  }

  /**
   * Mirrors the multi-algorithm heightmap generation
   * Uses Babylon.js PerlinNoise or manual Simplex implementation
   */
  private generateHeightmap(cx: number, cz: number, params: TerrainParams): Float32Array {
    const size = params.resolution;
    const heightmap = new Float32Array(size * size);

    for (let z = 0; z < size; z++) {
      for (let x = 0; x < size; x++) {
        let height = params.baseHeight;

        // Fractal noise (mirrors octave loop in Unity's PerlinNoise algorithm)
        let freq = params.frequency;
        let amp = params.amplitude;
        for (let o = 0; o < params.octaves; o++) {
          const nx = (cx * size + x) * freq + params.seed;
          const nz = (cz * size + z) * freq + params.seed;
          // Use a Perlin noise function (e.g., from a lib or manual)
          const noiseVal = perlinNoise2D(nx, nz); // Returns -1 to 1
          height += noiseVal * amp;
          freq *= params.lacunarity;
          amp *= params.persistence;
        }

        heightmap[z * size + x] = height * params.heightScale;
      }
    }

    return heightmap;
  }

  /** Mirrors GenerateTerrainMesh — vertex/UV/triangle assembly */
  private createMeshFromHeightmap(heightmap: Float32Array, params: TerrainParams): Mesh {
    // Use Babylon's VertexData API to build custom mesh from heightmap
    // Similar to Unity's mesh.vertices / mesh.triangles / mesh.uv assignment
    return MeshBuilder.CreateGround("terrain", {
      width: params.resolution,
      height: params.resolution,
      subdivisions: params.resolution - 1,
      updatable: true
    }, this.scene);
  }
}
```

```typescript
  slots: InventorySlot[];
  onChange?: () => void;  // Mirrors OnInventoryChanged event

  constructor() {
    this.slots = Array.from({ length: this.maxSlots }, () => ({
      item: null, amount: 0
    }));
  }

  /**
   * Mirrors AddItem — tries existing stacks first, then empty slots
   * Returns amount that couldn't be added (overflow)
   */
  addItem(item: ItemDef, amount: number): number {
    let remaining = amount;

    // Try existing stacks (mirrors Unity's foreach slot check)
    for (const slot of this.slots) {
      if (remaining <= 0) break;
      if (slot.item?.id === item.id && slot.amount < item.maxStackSize) {
        const canAdd = Math.min(remaining, item.maxStackSize - slot.amount);
        slot.amount += canAdd;
        remaining -= canAdd;
      }
    }

    // Try empty slots
    for (const slot of this.slots) {
      if (remaining <= 0) break;
      if (slot.item === null) {
        const canAdd = Math.min(remaining, item.maxStackSize);
        slot.item = item;
        slot.amount = canAdd;
        remaining -= canAdd;
      }
    }

    this.onChange?.();
    return remaining;
  }

  removeItem(itemId: string, amount: number): number {
    let toRemove = amount;
    for (const slot of this.slots) {
      if (toRemove <= 0) break;
      if (slot.item?.id === itemId) {
        const removed = Math.min(toRemove, slot.amount);
        slot.amount -= removed;
        toRemove -= removed;
        if (slot.amount <= 0) { slot.item = null; slot.amount = 0; }
      }
    }
    this.onChange?.();
    return toRemove;
  }
}
```

```typescript
// CellShader.ts
import { ShaderMaterial, Scene, Effect, Color3 } from "@babylonjs/core";

/** Register custom GLSL for toon/cel shading */
export function createCellShadedMaterial(scene: Scene, name: string, baseColor: Color3): ShaderMaterial {

  // Vertex Shader
  Effect.ShadersStore["cellVertexShader"] = `
    precision highp float;
    attribute vec3 position;
    attribute vec3 normal;
    uniform mat4 worldViewProjection;
    uniform mat4 world;
    varying vec3 vNormal;
    varying vec3 vWorldPos;

    void main() {
      gl_Position = worldViewProjection * vec4(position, 1.0);
      vNormal = normalize((world * vec4(normal, 0.0)).xyz);
      vWorldPos = (world * vec4(position, 1.0)).xyz;
    }
  `;

  // Fragment Shader — 3-band cel shading with rim light
  Effect.ShadersStore["cellFragmentShader"] = `
    precision highp float;
    varying vec3 vNormal;
    varying vec3 vWorldPos;
    uniform vec3 lightDirection;
    uniform vec3 cameraPosition;
    uniform vec3 baseColor;
    uniform vec3 rimColor;

    void main() {
      // Cel-shade bands
      float NdotL = dot(vNormal, normalize(-lightDirection));
      float intensity;
      if (NdotL > 0.7) intensity = 1.0;
      else if (NdotL > 0.35) intensity = 0.6;
      else if (NdotL > 0.05) intensity = 0.3;
      else intensity = 0.15;

      vec3 color = baseColor * intensity;

      // Rim lighting (anime edge glow)
      vec3 viewDir = normalize(cameraPosition - vWorldPos);
      float rim = 1.0 - max(dot(viewDir, vNormal), 0.0);
      rim = smoothstep(0.6, 1.0, rim);
      color += rimColor * rim * 0.5;

      gl_FragColor = vec4(color, 1.0);
    }
  `;

  const mat = new ShaderMaterial(name, scene, "cell", {
    attributes: ["position", "normal"],
    uniforms: ["worldViewProjection", "world", "lightDirection",
               "cameraPosition", "baseColor", "rimColor"]
  });

  mat.setVector3("lightDirection", new BABYLON.Vector3(-1, -2, -1));
  mat.setColor3("baseColor", baseColor);
  mat.setColor3("rimColor", new Color3(0, 1, 1)); // Cyan rim

  // Update camera position each frame
  scene.registerBeforeRender(() => {
    mat.setVector3("cameraPosition", scene.activeCamera!.position);
  });

  return mat;
}
```

### Common Patterns

```
// UNITY C#                          → BABYLON.JS TYPESCRIPT
//──────────────────────────────────────────────────────────────
MonoBehaviour                        → class + registerBeforeRender()
[SerializeField] float x = 5f       → config interface { x: number } // default 5
Update() { }                         → scene.registerBeforeRender(() => { })
Time.deltaTime                       → engine.getDeltaTime() / 1000
Vector3.Distance(a, b)               → Vector3.Distance(a, b)
Vector3.Lerp(a, b, t)                → Vector3.Lerp(a, b, t)
Quaternion.Slerp(a, b, t)            → Quaternion.Slerp(a, b, t, result)
Quaternion.LookRotation(dir)         → mesh.lookAt(targetPos)
transform.position                   → mesh.position
transform.forward                    → mesh.forward (or getDirection)
transform.rotation                   → mesh.rotationQuaternion
Physics.Raycast(origin, dir)         → scene.pickWithRay(new Ray(...))
Physics.OverlapBox(center, half)     → mesh.intersectsMesh(other) or custom AABB
Rigidbody.AddForce(f, Impulse)       → physicsBody.applyImpulse(f, contactPoint)
NavMeshAgent.SetDestination(p)       → crowd.agentGoto(agentIdx, navPlugin.getClosestPoint(p))
Instantiate(prefab)                  → mesh.clone() or mesh.createInstance()
Destroy(obj)                         → mesh.dispose()
Destroy(obj, delay)                  → setTimeout(() => mesh.dispose(), delay * 1000)
Invoke(method, delay)                → setTimeout(() => this.method(), delay * 1000)
StartCoroutine(routine)              → async/await or Observable patterns
Animator.CrossFade("Anim", 0.1f)     → animationGroup.start(loop, speed, from, to)
GetComponent<T>()                    → stored reference or mesh.metadata
CompareTag("Enemy")                  → Tags.MatchesQuery(mesh, "enemy")
FindGameObjectsWithTag("Enemy")      → scene.getMeshesByTags("enemy")
Input.GetKeyDown(KeyCode.Alpha1)     → scene.onKeyboardObservable, check key === "1"
ScriptableObject                     → JSON file or TypeScript interface
PlayerPrefs                          → localStorage / IndexedDB
AudioSource.PlayOneShot(clip)        → new Sound("name", url, scene).play()
CanvasGroup.alpha                    → React state + CSS opacity
```
